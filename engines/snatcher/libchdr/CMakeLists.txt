cmake_minimum_required(VERSION 3.20)
project(snatcher_chdr LANGUAGES C)
file(GLOB SNATCHER_CHDR_SOURCES CONFIGURE_DEPENDS src/*.c)
list(REMOVE_ITEM SNATCHER_CHDR_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/libchdr_huffman.c")
add_library(snatcher-chdr STATIC ${SNATCHER_CHDR_SOURCES} ../chd_huffman.c ../chd_miniz.c deps/lzma-25.01/src/LzmaDec.c deps/zstd-1.5.7/zstddeclib.c)
target_include_directories(snatcher-chdr PUBLIC include)
target_compile_definitions(snatcher-chdr PRIVATE MINIZ_NO_ARCHIVE_APIS MINIZ_NO_DEFLATE_APIS MINIZ_NO_STDIO MINIZ_NO_TIME WANT_RAW_DATA_SECTOR=1 WANT_SUBCODE=1 VERIFY_BLOCK_CRC=1)
if(MSVC)
 target_compile_definitions(snatcher-chdr PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(snatcher-chdr PRIVATE /we4456)
endif()
