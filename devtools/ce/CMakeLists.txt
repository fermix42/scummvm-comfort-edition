cmake_minimum_required(VERSION 3.16)
project(ce_native_regressions LANGUAGES CXX)
enable_testing()
add_executable(sq5_battle_cheats test_sq5_battle_cheats.cpp)
target_include_directories(sq5_battle_cheats PRIVATE ../.. ../../engines)
target_compile_definitions(sq5_battle_cheats PRIVATE WIN32)
# Assertions are the checks in this existing regression executable.
target_compile_options(sq5_battle_cheats PRIVATE /UNDEBUG)
add_test(NAME sq5_battle_cheats COMMAND sq5_battle_cheats)
add_executable(msvc_warning_contract test_msvc_warning_contract.cpp)
target_compile_options(msvc_warning_contract PRIVATE /UNDEBUG)
add_test(NAME msvc_warning_contract COMMAND msvc_warning_contract)

# Both configurations exercise the actual GL macro policy with mock callbacks.
foreach(mode IN ITEMS release debug)
    add_executable(gl_polling_${mode} test_gl_polling_policy.cpp)
    target_include_directories(gl_polling_${mode} PRIVATE ../..)
    if(mode STREQUAL "release")
        target_compile_definitions(gl_polling_${mode} PRIVATE RELEASE_BUILD EXPECT_ROUTINE_POLLING=0)
    else()
        target_compile_definitions(gl_polling_${mode} PRIVATE EXPECT_ROUTINE_POLLING=1)
    endif()
    add_test(NAME gl_polling_${mode} COMMAND gl_polling_${mode})
endforeach()
